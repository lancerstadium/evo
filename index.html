    3 Demo

    • Here is a c demo for libevo :
    #include <evo.h>

int main() {
    // ...
    serializer_t * sez = serializer_get("onnx");
    model_t * mdl = sez->load_file(sez, "model/mnist_8/model.onnx");
    tensor_t * t1 = model_get_tensor(mdl, "Input3");
    tensor_dump(t1);
    // ...
    return 0;
}
